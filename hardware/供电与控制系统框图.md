# 供电与控制系统框图
```mermaid
flowchart LR
    BAT["2S航模电池 450mAh 75C"]
    SW["总开关"]
    SPLIT["分路"]

    %% 动力支路：负压泵无刷电调
    ESC["12A无刷电调"]
    FAN["1104无刷电机<br/>负压泵4300kV"]

    %% 控制支路：降压到5V母线
    BUCK["降压板 7.4V → 5V"]
    BUS5V["5V母线"]

    %% 行走电机驱动
    DRV["DRV8833电机驱动 VM=5V"]
    M1["N20 6V电机（左轮）"]
    M2["N20 6V电机（右轮）"]

    %% 主控与传感器
    ESP["ESP32-S3开发板"]
    IMU["IMU MPU6050"]

    %% 停机逻辑
    ALARM{"停机触发判断"}
    CUT["切断动力输出"]
    GND["【公共GND】所有模块GND统一接入"]

    %% 主回路
    BAT --> SW
    SW --> SPLIT

    %% 动力支路（负压风机，7.4V直供）
    SPLIT --> ESC
    ESC --> FAN

    %% 控制支路（降压5V母线）
    SPLIT --> BUCK
    BUCK --> BUS5V

    %% 5V母线负载：DRV8833 + 主控+传感器
    BUS5V --> DRV
    DRV --> M1
    DRV --> M2
    BUS5V --> ESP

    %% 信号线
    ESP -->|PWM信号| ESC
    ESP -->|方向控制信号| DRV
    M1 -->|编码器A/B 3.3V| ESP
    M2 -->|编码器A/B 3.3V| ESP
    IMU -->|I2C SDA/SCL| ESP

    %% 停机判断：网页/遥控器急停 + 硬件总开关
    ESP --> ALARM
    ALARM -- 远程急停按钮按下 --> CUT

    %% 共地
    ESC --- GND
    DRV --- GND
    ESP --- GND
    IMU --- GND
